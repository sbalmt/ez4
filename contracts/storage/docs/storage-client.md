# EZ4: Storage Client

The storage client exposes a simple, type-safe object storage API for the bucket service. It is generated by EZ4 and injected into contexts that have access to the storage service.

## Client API

The client API provides a unified, provider-agnostic way to access object storage no matter which provider is being used.

#### Get object stats

```ts
export async function anotherHandler(_request: any, { myStorage }: Service.Context<MyService>) {
  const stats = await myStorage.stat('path/file.txt');

  if (stats) {
    console.log(stats.type, stats.size, stats.metadata);
  }
}
```

> Returns object metadata or `undefined` when the object does not exist.

#### Check object existence

```ts
export async function anotherHandler(_request: any, { myStorage }: Service.Context<MyService>) {
  const exists = await myStorage.exists('path/file.txt');
}
```

> Returns `true` when the object exists in the bucket.

#### Write an object

```ts
export async function anotherHandler(_request: any, { myStorage }: Service.Context<MyService>) {
  await myStorage.write('path/file.txt', 'Hello storage', {
    contentType: 'text/plain',
    metadata: {
      source: 'docs'
    },
    headers: {
      cacheControl: 'public, max-age=3600',
      expires: new Date('1991-04-23')
    }
  });
}
```

> The content can be a string, `Uint8Array`, `Buffer`, or readable stream.

#### Read an object

```ts
export async function anotherHandler(_request: any, { myStorage }: Service.Context<MyService>) {
  const content = await myStorage.read('path/file.txt');
}
```

> Returns the object contents as a `Buffer`.

#### Delete an object

```ts
export async function anotherHandler(_request: any, { myStorage }: Service.Context<MyService>) {
  await myStorage.delete('path/file.txt');
}
```

#### Copy an object

```ts
export async function anotherHandler(_request: any, { myStorage }: Service.Context<MyService>) {
  await myStorage.copy('path/file.txt', 'archives/file.txt');
}
```

#### Scan objects

```ts
export async function anotherHandler(_request: any, { myStorage }: Service.Context<MyService>) {
  for await (const object of myStorage.scan('path/')) {
    console.log(object.key, object.size, object.modifiedAt);
  }
}
```

> When no prefix is provided, the scan includes all objects visible to the bucket client.

#### Get a signed stats URL

```ts
export async function anotherHandler(_request: any, { myStorage }: Service.Context<MyService>) {
  const url = await myStorage.getStatUrl('path/file.txt', {
    expiresIn: 300
  });
}
```

#### Get a signed write URL

```ts
export async function anotherHandler(_request: any, { myStorage }: Service.Context<MyService>) {
  const url = await myStorage.getWriteUrl('path/upload.txt', {
    expiresIn: 300,
    contentType: 'text/plain',
    metadata: {
      source: 'browser'
    },
    headers: {
      cacheControl: 'no-cache'
    }
  });
}
```

#### Get a signed read URL

```ts
export async function anotherHandler(_request: any, { myStorage }: Service.Context<MyService>) {
  const url = await myStorage.getReadUrl('path/file.txt', {
    expiresIn: 300
  });
}
```
